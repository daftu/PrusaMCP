cmake_minimum_required(VERSION 3.20)
project(prusamcp_preset_core LANGUAGES C CXX)
set(PRUSASLICER_SOURCE_DIR "" CACHE PATH "PrusaSlicer source at b028299c770b8380ee81c921a2867d522f288123")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${PRUSASLICER_SOURCE_DIR}" OUTPUT_VARIABLE SOURCE_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT SOURCE_REV STREQUAL "b028299c770b8380ee81c921a2867d522f288123")
  message(FATAL_ERROR "PrusaSlicer source must match pinned 2.9.6 revision")
endif()
set(CMAKE_CXX_STANDARD 17)
set(Boost_USE_STATIC_LIBS ON)
find_package(Boost 1.83 REQUIRED COMPONENTS filesystem locale log regex thread nowide)
find_package(Eigen3 REQUIRED)
find_package(TBB REQUIRED)
find_package(LibBGCode REQUIRED COMPONENTS Binarize)
find_package(nlohmann_json REQUIRED)
find_package(JPEG REQUIRED)
set(S "${PRUSASLICER_SOURCE_DIR}/src/libslic3r")
set(B "${PRUSASLICER_SOURCE_DIR}/bundled_deps")
set(SLIC3R_APP_NAME PrusaSlicer)
set(SLIC3R_APP_KEY PrusaSlicer)
set(SLIC3R_VERSION 2.9.6)
set(SLIC3R_BUILD_ID "PrusaMCP-native-${SOURCE_REV}")
configure_file("${S}/libslic3r_version.h.in" "${CMAKE_CURRENT_BINARY_DIR}/libslic3r_version.h" @ONLY)
add_executable(prusamcp-preset-core main.cpp
  "${S}/Config.cpp" "${S}/PrintConfig.cpp" "${S}/Preset.cpp" "${S}/PresetBundle.cpp"
  "${S}/PlaceholderParser.cpp" "${S}/CustomParametersHandling.cpp" "${S}/AppConfig.cpp"
  "${S}/utils.cpp" "${S}/Time.cpp" "${S}/Platform.cpp" "${S}/Semver.cpp"
  "${S}/GCode/Thumbnails.cpp" "${S}/Flow.cpp" "${S}/Point.cpp"
  "${B}/localesutils/LocalesUtils.cpp" "${B}/semver/semver.c")
target_include_directories(prusamcp-preset-core PRIVATE "${PRUSASLICER_SOURCE_DIR}/src" "${S}" "${CMAKE_CURRENT_BINARY_DIR}" "${B}/localesutils" "${B}/semver" "${B}/fast_float" "${B}/qoi" "${B}/miniz" "${B}/admesh" "${B}/ankerl" "${B}/tcbspan" "${B}/int128")
target_link_libraries(prusamcp-preset-core PRIVATE ${Boost_LIBRARIES} Eigen3::Eigen TBB::tbb TBB::tbbmalloc LibBGCode::bgcode_binarize nlohmann_json::nlohmann_json JPEG::JPEG)
target_compile_options(prusamcp-preset-core PRIVATE -ffunction-sections -fdata-sections)
if(APPLE)
  target_link_options(prusamcp-preset-core PRIVATE -Wl,-dead_strip)
  target_link_libraries(prusamcp-preset-core PRIVATE iconv)
else()
  target_link_options(prusamcp-preset-core PRIVATE -Wl,--gc-sections)
endif()
install(TARGETS prusamcp-preset-core RUNTIME DESTINATION bin)
